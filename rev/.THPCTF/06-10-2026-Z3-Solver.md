beberapa bisa di hitung manual atau menggunakan skrip python, namun langkah terbaik untuk menangani kasus persamaan seperti ini adalah menggunakan skrip python + lib z3 Solver.

```python
from z3 import *
from pwn import *

global key
key = []

def l1():
    print("[+] l1 : sat")
    key.append(str(18).encode()) # hasil dekompilasi ghidra sudah tertulis 18. asumsi ini merupakan free key , atau memang sebelumnya ada persamaan namun sudah di selesaikan oleh dekompiler ghidra sendiri.

def l2():
    s = Solver()
    a,b = BitVecs('a b',32)
    s.add(b+a*2 == 20 , a-b ==1)
    if s.check() == sat:
        print("[+] l2 : sat")
        key.append(str(s.model()[a].as_long()).encode())
        key.append(str(s.model()[b].as_long()).encode())
    else:
        print("[-] l2 : usat")

def l3():
    s = Solver()
    p,q,r = BitVecs('p q r',32)
    s.add(r + p + q == 24 , p * 3 == q , p + q == r)
    if s.check() == sat:
        print("[+] l3 : sat")
        key.append(str(s.model()[p].as_long()).encode())
        key.append(str(s.model()[q].as_long()).encode())
        key.append(str(s.model()[r].as_long()).encode())
    else:
        print("[-] l3 : usat")
        return None, None, None
   
def l4():
    s = Solver()
    t = BitVec('t',32)
    s.add(ULT(t,256)) # unsigned less than, digunakan untuk perbandingan kurang dari jika angka nya bertipe data unsigned int
    s.add(Extract(7,0,t) == -1) # mengambil tepat 8 bit terakhir (least significant bit) (1 byte) untuk dibandingkan dengan -1
    if s.check() == sat:
        print("[+] l4 : sat")
        key.append(str(s.model()[t].as_long()).encode())
    else:
        print("[-] l4 : usat")
        return None

def l5():
    s = Solver()
    k = BitVec('k',32)
    s.add(LShR(k,4) == 10 , k == 167) # Logical Shift Right, pada operasi pergeseran ini, nilai bit paling kiri akan di isi oleh nol. cocok untuk tipe data unsigned 
    if s.check() == sat: 
        print("[+] l5 : sat")
        key.append(str(s.model()[k].as_long()).encode())
    else:
        print("[-] l5 : usat")
        return None


l1();l2();l3();l4();l5()

i = 0
p = process("./level2")
while True:
    try:
        K = key[i].decode()
        print("KEY FOUND : " ,text.bold_green(K))
        p.sendline(key[i])
        i += 1
    except:
        print()
        break

print(p.recv().decode())
p.close()
# semua tipe data di solver di deklarasikan dengan BitVec(x,32), saya menggunakan 32 bit karena semua tipe data yang ditampilkan hasil dekompilasi adalah uint
# dimana int itu ukurannya adalah 4 byte, (32 bit)
```
